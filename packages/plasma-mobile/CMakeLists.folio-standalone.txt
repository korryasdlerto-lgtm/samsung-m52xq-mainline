# m52xq: only the Folio homescreen applet of plasma-mobile 6.7.5 (patch 0005-m52xq-folio-favourites-right-in-landscape),
# against the phone's exact versions (Qt 6.11.1, KF 6.30, Plasma 6.7.5)
cmake_minimum_required(VERSION 3.16)
project(plasma-mobile)
set(PROJECT_VERSION "6.7.5")
set(QT_MIN_VERSION "6.10.0")
set(KF6_MIN_VERSION "6.26.0")
set(KDE_COMPILERSETTINGS_LEVEL "5.82")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ECM ${KF6_MIN_VERSION} REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
set(KF_IGNORE_PLATFORM_CHECK ON CACHE BOOL "Don't check platform that is being built on")
include(KDEInstallDirs)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)
include(ECMQtDeclareLoggingCategory)
include(ECMQmlModule)
add_definitions(-DQT_NO_URL_CAST_FROM_STRING)
find_package(Qt6 ${QT_MIN_VERSION} CONFIG REQUIRED COMPONENTS Core Gui Qml Quick)
find_package(KF6 ${KF6_MIN_VERSION} REQUIRED COMPONENTS I18n Service KIO Notifications WindowSystem JobWidgets Config)
find_package(Plasma CONFIG REQUIRED)
find_package(PlasmaQuick CONFIG REQUIRED)
find_package(KWayland CONFIG REQUIRED)
add_subdirectory(src/containments/homescreens/folio)
